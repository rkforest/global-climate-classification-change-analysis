Objective: Transform the acquired ERA5 temperature and precipitation data into analysis-ready monthly climatologies for the REF, CSN, and RCI climate intervals. This notebook creates reusable raster caches, calculates twelve climatological monthly values for each climate variable and interval, integrates the shared spatial layers, and saves the transformed datasets for subsequent determinant calculation.

Each climate interval has its own ERA5 NetCDF temperature and precipitation files. This one-period-per-source design prioritizes transparency and makes annual updates to the Recent Climate Period independent of the two stable WMO periods.

Because repeated grouped reads from the compressed NetCDF files are slow, each source file is converted once to an uncompressed GeoTIFF processing cache. Subsequent climatological aggregation is performed from these cache files with terra. The cache is entirely reproducible and may be deleted at any time. A cache file is rebuilt automatically when it is missing or older than its corresponding NetCDF source file.

Only the twelve-layer climatologies are converted to data frames for Parquet output.

fn_read_era5 <- function(file_path, variable, require_time = TRUE) {
  # CDS NetCDF files may expose variables as GDAL subdatasets.
  r <- try(rast(file_path, subds = variable), silent = TRUE)

  if (inherits(r, "try-error")) {
    r <- rast(file_path)
  }

  if (require_time && !has.time(r)) {
    stop("No time dimension was detected in: ", file_path)
  }

  # Keep the native ERA5 0..360 longitude geometry while processing the
  # hundreds of monthly layers. Rotation is deferred until after aggregation.
  r
}

fn_source_row <- function(climate_variable, period_id) {
  candidate <- source_files |>
    filter(
      .data$climate_variable == .env$climate_variable,
      .data$period_id == .env$period_id
    )

  if (nrow(candidate) != 1) {
    stop(
      "Expected exactly one source file for variable ", climate_variable,
      " and period ", period_id, "."
    )
  }

  candidate
}

fn_ensure_cache <- function(source_row) {
  source_path <- path(dir_raw, source_row$file_name[[1]])
  cache_path <- path(dir_cache, source_row$cache_name[[1]])

  if (!file_exists(source_path)) {
    stop("Missing source NetCDF file: ", source_path)
  }

  rebuild <- !file_exists(cache_path)

  if (!rebuild) {
    source_mtime <- file_info(source_path)$modification_time
    cache_mtime <- file_info(cache_path)$modification_time
    rebuild <- cache_mtime < source_mtime
  }

  start_time <- Sys.time()

  if (rebuild) {
    r <- fn_read_era5(
      source_path,
      source_row$netcdf_variable[[1]]
    )

    writeRaster(
      r,
      cache_path,
      overwrite = TRUE,
      datatype = "FLT4S",
      filetype = "GTiff",
      gdal = c("COMPRESS=NONE", "BIGTIFF=IF_SAFER")
    )

    rm(r)
    cache_status <- "created"
  } else {
    cache_status <- "reused"
  }

  elapsed_seconds <- as.numeric(
    difftime(Sys.time(), start_time, units = "secs")
  )

  tibble(
    period_id = source_row$period_id[[1]],
    period_abbr = source_row$period_abbr[[1]],
    source_file = source_row$file_name[[1]],
    cache_file = source_row$cache_name[[1]],
    cache_status = cache_status,
    cache_size_gib = round(as.numeric(file_info(cache_path)$size) / 1024^3, 3),
    elapsed_seconds = round(elapsed_seconds, 1)
  )
}

fn_read_cache <- function(source_row) {
  cache_path <- path(dir_cache, source_row$cache_name[[1]])

  if (!file_exists(cache_path)) {
    stop("Missing raster cache: ", cache_path)
  }

  r <- rast(cache_path)

  expected_layers <-
    (source_row$end[[1]] - source_row$beg[[1]] + 1) * 12

  if (nlyr(r) != expected_layers) {
    stop(
      "Expected ", expected_layers, " layers in ", cache_path,
      " but found ", nlyr(r), "."
    )
  }

  # GeoTIFF does not reliably preserve the ERA5 NetCDF time dimension.
  # Restore the known complete monthly sequence from the period metadata.
  time(r) <- seq(
    as.Date(sprintf("%d-01-01", source_row$beg[[1]])),
    as.Date(sprintf("%d-12-01", source_row$end[[1]])),
    by = "month"
  )

  r
}

fn_spatial_layers <- function(template) {
  mask_path <- path(dir_raw, mask_file)

  if (!file_exists(mask_path)) {
    stop(
      "Missing ", mask_file,
      ". Run the data-acquisition notebook to download the mask on ",
      "the native ERA5 grid."
    )
  }

  lsm <- fn_read_era5(mask_path, "lsm", require_time = FALSE)[[1]]
  names(lsm) <- "lsm"

  if (!compareGeom(template[[1]], lsm, stopOnError = FALSE)) {
    stop(
      "The climate raster and land-sea mask are not on the same grid. ",
      "Do not resample the mask; reacquire it on the native ERA5 grid."
    )
  }

  # Ellipsoidal lon/lat grid-cell area, expressed in million km² to preserve
  # the schema and units used by the original project.
  area <- cellSize(template[[1]], unit = "km") / 1e6
  names(area) <- "area"

  list(lsm = lsm, area = area)
}

fn_aggregate_temperature <- function(r) {
  month_index <- as.integer(format(as.Date(time(r)), "%m"))

  out <- tapp(
    r,
    index = month_index,
    fun = "mean",
    na.rm = TRUE
  ) - 273.15

  names(out) <- paste0("m", 1:12)
  out
}

fn_aggregate_precipitation <- function(r) {
  dates <- as.Date(time(r), tz = "UTC")

  # ERA5 monthly-mean total precipitation is a mean daily accumulation.
  # Convert each year-month to its actual monthly total in millimetres before
  # averaging by calendar month. This handles leap-year Februaries exactly.
  month_start <- as.Date(format(dates, "%Y-%m-01"))
  next_month_start <- as.Date(format(month_start + 32, "%Y-%m-01"))
  days_in_month <- as.numeric(next_month_start - month_start)

  monthly_total_mm <- r * days_in_month * 1000
  month_index <- as.integer(format(dates, "%m"))

  out <- tapp(
    monthly_total_mm,
    index = month_index,
    fun = "mean",
    na.rm = TRUE
  )

  names(out) <- paste0("m", 1:12)
  out
}

fn_raster_to_parquet <- function(climate_raster, spatial, output_path) {
  if (!compareGeom(climate_raster[[1]], spatial$lsm, stopOnError = FALSE)) {
    stop("Climate raster and land-sea mask grids do not match.")
  }

  combined <- c(spatial$area, spatial$lsm, climate_raster)

  # Rotate only the reduced 14-layer stack (area + mask + 12 climatology
  # months), rather than hundreds of raw ERA5 monthly layers.
  if (xmax(combined) > 180) {
    combined <- rotate(combined)
  }

  df <- as.data.frame(
    combined,
    xy = TRUE,
    na.rm = FALSE
  ) |>
    rename(lon = x, lat = y) |>
    select(lat, lon, area, lsm, everything())

  write_parquet(df, output_path)
  invisible(NULL)
}

fn_process_dataset <- function(climate_variable, period_id, spatial) {
  source <- fn_source_row(climate_variable, period_id)
  start_time <- Sys.time()

  r <- fn_read_cache(source)

  aggregated <- switch(
    climate_variable,
    t = fn_aggregate_temperature(r),
    p = fn_aggregate_precipitation(r),
    stop("Unknown climate variable: ", climate_variable)
  )

  output_file <- sprintf(
    "%s_agg_%d-%d.parquet",
    climate_variable,
    source$beg[[1]],
    source$end[[1]]
  )
  output_path <- path(dir_transformed, output_file)

  fn_raster_to_parquet(aggregated, spatial, output_path)

  elapsed_seconds <- as.numeric(
    difftime(Sys.time(), start_time, units = "secs")
  )

  tibble(
    period_id = source$period_id[[1]],
    period_name = source$period_name[[1]],
    period_label = source$period_label[[1]],
    period_abbr = source$period_abbr[[1]],
    output_file = output_file,
    source_file = source$file_name[[1]],
    cache_file = source$cache_name[[1]],
    variable_name = source$netcdf_variable[[1]],
    period = sprintf("%d-%d", source$beg[[1]], source$end[[1]]),
    latitude_count = nrow(r),
    longitude_count = ncol(r),
    input_month_count = nlyr(r),
    input_values_count = ncell(r) * nlyr(r),
    output_month_count = nlyr(aggregated),
    output_values_count = ncell(aggregated) * nlyr(aggregated),
    elapsed_seconds = round(elapsed_seconds, 1)
  )
}

if (run_transformation) {

  # Check or build the GeoTIFF processing caches.
  cache_log <- bind_rows(
    lapply(
      seq_len(nrow(source_files)),
      function(i) fn_ensure_cache(source_files[i, ])
    )
  )

  # Establish the common spatial geometry from the reference-period
  # temperature raster.
  reference_source <- fn_source_row(
    "t",
    "reference_period"
  )

  template_source <- fn_read_cache(
    reference_source
  )

  spatial <- fn_spatial_layers(
    template_source[[1]]
  )

  rm(template_source)

  # Transform temperature and precipitation for all climate intervals.
  log_rows <- list()
  row_id <- 1L

  for (period_id in climate_periods$period_id) {
    for (climate_variable in c("t", "p")) {

      log_rows[[row_id]] <- fn_process_dataset(
        climate_variable = climate_variable,
        period_id = period_id,
        spatial = spatial
      )

      row_id <- row_id + 1L
    }
  }

  log_df <- bind_rows(log_rows)

  # Preserve processing metadata so the notebook can subsequently be rendered
  # without repeating the expensive transformation.
  readr::write_csv(
    cache_log,
    cache_log_path
  )

  readr::write_csv(
    log_df,
    transformation_log_path
  )

} else {

  # Presentation-only mode: use metadata from the most recent successful
  # transformation rather than rebuilding caches or transformed datasets.
  required_logs <- c(
    cache_log_path,
    transformation_log_path
  )

  if (!all(file_exists(required_logs))) {
    stop(
      "Transformation logs are missing. ",
      "Set run_transformation <- TRUE and run the notebook once."
    )
  }

  cache_log <- readr::read_csv(
    cache_log_path,
    show_col_types = FALSE
  )

  log_df <- readr::read_csv(
    transformation_log_path,
    show_col_types = FALSE
  )
}

```{r}
  out
}

fn_raster_to_parquet <- function(climate_raster, spatial, output_path) {
  if (!compareGeom(climate_raster[[1]], spatial$lsm, stopOnError = FALSE)) {
    stop("Climate raster and land-sea mask grids do not match.")
  }

  combined <- c(spatial$area, spatial$lsm, climate_raster)

  # Rotate only the reduced 14-layer stack (area + mask + 12 climatology
  # months), rather than hundreds of raw ERA5 monthly layers.
  if (xmax(combined) > 180) {
    combined <- rotate(combined)
  }

  df <- as.data.frame(
    combined,
    xy = TRUE,
    na.rm = FALSE
  ) |>
    rename(lon = x, lat = y) |>
    select(lat, lon, area, lsm, everything())

  write_parquet(df, output_path)
  invisible(NULL)
}

fn_process_dataset <- function(climate_variable, period_id, spatial) {
  source <- fn_source_row(climate_variable, period_id)
  start_time <- Sys.time()

  r <- fn_read_cache(source)

  aggregated <- switch(
    climate_variable,
    t = fn_aggregate_temperature(r),
    p = fn_aggregate_precipitation(r),
    stop("Unknown climate variable: ", climate_variable)
  )

  output_file <- sprintf(
    "%s_agg_%d-%d.parquet",
    climate_variable,
    source$beg[[1]],
    source$end[[1]]
  )
  output_path <- path(dir_transformed, output_file)

  fn_raster_to_parquet(aggregated, spatial, output_path)

  elapsed_seconds <- as.numeric(
    difftime(Sys.time(), start_time, units = "secs")
  )

  tibble(
    period_id = source$period_id[[1]],
    period_name = source$period_name[[1]],
    period_label = source$period_label[[1]],
    period_abbr = source$period_abbr[[1]],
    output_file = output_file,
    source_file = source$file_name[[1]],
    cache_file = source$cache_name[[1]],
    variable_name = source$netcdf_variable[[1]],
    period = sprintf("%d-%d", source$beg[[1]], source$end[[1]]),
    latitude_count = nrow(r),
    longitude_count = ncol(r),
    input_month_count = nlyr(r),
    input_values_count = ncell(r) * nlyr(r),
    output_month_count = nlyr(aggregated),
    output_values_count = ncell(aggregated) * nlyr(aggregated),
    elapsed_seconds = round(elapsed_seconds, 1)
  )
}
```

## Main Process

```{r}
#| label: main-process
#| results: hide

if (run_transformation) {

  # Check or build the GeoTIFF processing caches.
  cache_log <- bind_rows(
    lapply(
      seq_len(nrow(source_files)),
      function(i) fn_ensure_cache(source_files[i, ])
    )
  )

  # Establish the common spatial geometry from the reference-period
  # temperature raster.
  reference_source <- fn_source_row(
    "t",
    "reference_period"
  )

  template_source <- fn_read_cache(
    reference_source
  )

  spatial <- fn_spatial_layers(
    template_source[[1]]
  )

  rm(template_source)

  # Transform temperature and precipitation for all climate intervals.
  log_rows <- list()
  row_id <- 1L

  for (period_id in climate_periods$period_id) {
    for (climate_variable in c("t", "p")) {

      log_rows[[row_id]] <- fn_process_dataset(
        climate_variable = climate_variable,
        period_id = period_id,
        spatial = spatial
      )

      row_id <- row_id + 1L
    }
  }

  log_df <- bind_rows(log_rows)

  # Preserve processing metadata so the notebook can subsequently be rendered
  # without repeating the expensive transformation.
  readr::write_csv(
    cache_log,
    cache_log_path
  )

  readr::write_csv(
    log_df,
    transformation_log_path
  )

} else {

  # Presentation-only mode: use metadata from the most recent successful
  # transformation rather than rebuilding caches or transformed datasets.
  required_logs <- c(
    cache_log_path,
    transformation_log_path
  )

  if (!all(file_exists(required_logs))) {
    stop(
      "Transformation logs are missing. ",
      "Set run_transformation <- TRUE and run the notebook once."
    )
  }

  cache_log <- readr::read_csv(
    cache_log_path,
    show_col_types = FALSE
  )

  log_df <- readr::read_csv(
    transformation_log_path,
    show_col_types = FALSE
  )
}
```
